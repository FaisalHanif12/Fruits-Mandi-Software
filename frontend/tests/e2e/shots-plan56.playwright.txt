// Screenshots of the first screen of a new install and the restore pages (plan 56) in both languages
// at 1024x600 and 1366x768, taken from the development page /dev/restore with the Playwright tool. It
// is a developer tool: run it with the Vite dev server up (`npx vite --port 5196`). It also measures
// sideways overflow (NFR-USE-15): the page, and every box inside it, must not scroll sideways.
// Usage with the Playwright MCP tool: browser_run_code_unsafe with this code. (It is a snippet for
// that tool, not a module, so it is kept as .txt and the linter does not read it.)
async (page) => {
  const OUT =
    '/Users/faisalhanif/MyProfile/My Projects /Fruit Mandi Parallel/d5/frontend/tests/e2e/output/screenshots/plan56';
  const BASE = 'http://localhost:5196/#/dev/restore';
  const SIZES = [
    [1024, 600],
    [1366, 768],
  ];
  const STATES = [
    ['start', '', async () => {}],
    ['where-cloud', '?at=where', async () => {}],
    [
      'where-cloud-refused',
      '?at=wherefilled&down',
      async (p) => {
        // The language button had the click, so the first field takes the focus again.
        await p.locator('main input, input').first().focus();
        for (let i = 0; i < 6; i += 1) await p.keyboard.press('Enter');
        await p.waitForTimeout(400);
      },
    ],
    [
      'where-folder',
      '?at=where',
      async (p) => {
        // The second group is the place choice (the first one is the language switch).
        await p.locator('[role="group"]').nth(1).locator('button').nth(1).click();
        await p.waitForTimeout(200);
      },
    ],
    ['copy', '?at=copy', async () => {}],
    ['copy-many', '?at=copy&many', async () => {}],
    ['copy-empty', '?at=copy&empty', async () => {}],
    ['check', '?at=check', async () => {}],
    ['check-upgrade', '?at=check&upgrade', async () => {}],
    [
      'copy-damaged',
      '?at=copy&bad',
      async (p) => {
        await p.locator('input[type="password"]').first().focus();
        await p.keyboard.type('ABCD-EFGH-JKLM-NPQR-STUV-WXYZ');
        await p.keyboard.press('Enter');
        await p.waitForTimeout(500);
      },
    ],
    ['running', '?at=running', async () => {}],
    ['done', '?at=done', async () => {}],
    [
      'failed',
      '?at=check&fail',
      async (p) => {
        await p.keyboard.press('Enter');
        await p.waitForTimeout(300);
        await p.keyboard.press('Enter');
        await p.waitForTimeout(500);
      },
    ],
  ];
  const report = [];
  for (const language of ['ur', 'en']) {
    for (const [w, h] of SIZES) {
      await page.setViewportSize({ width: w, height: h });
      for (const [name, query, act] of STATES) {
        // A change of the hash alone keeps the page, so each state starts from a blank one.
        await page.goto('about:blank');
        await page.goto(`${BASE}${query}`);
        await page.waitForTimeout(500);
        if (language === 'en') {
          await page.getByRole('button', { name: 'English' }).first().click();
          await page.waitForTimeout(300);
        }
        await act(page);
        await page.waitForTimeout(300);
        await page.screenshot({ path: `${OUT}/${name}-${language}-${w}x${h}.png` });
        const wide = await page.evaluate(() => {
          const out = [];
          const root = document.documentElement;
          if (root.scrollWidth > root.clientWidth) out.push('page');
          for (const el of document.querySelectorAll('main *, section, [role="grid"], table')) {
            if (el.scrollWidth > el.clientWidth + 1 && getComputedStyle(el).overflowX !== 'visible') {
              out.push(el.tagName + '.' + String(el.className).slice(0, 40));
            }
          }
          return out;
        });
        if (wide.length > 0) report.push(`${name} ${language} ${w}x${h}: ${wide.join(' | ')}`);
      }
    }
  }
  return report.length === 0 ? 'no sideways overflow in any shot' : report;
}
