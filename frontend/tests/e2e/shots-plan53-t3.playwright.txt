// Screenshots of plan 53 T3 (PRD FR-10.11, FR-15.6, FR-7.61): the collection list page without a day
// (the date box now says "receipt date"), and the opening-date offer of the account form when the day
// of the opening is closed: today open (two buttons) and today closed too (one button). Both languages,
// at 1366 x 768 and 1024 x 600. A developer tool: run it with the Vite dev server up, as a snippet for
// `playwright-core` (kept as .txt so the linter does not read it). The pages are /dev/vouchers?collection
// and /dev/index?screen=account&closedopen|closednight (tests/helpers/dev-bridge-index.ts).
async (page) => {
  const OUT =
    '/Users/faisalhanif/MyProfile/My Projects /Fruit Mandi Parallel/d5/frontend/tests/e2e/output/screenshots/plan53';
  const BASE = 'http://localhost:5199/#/dev/';
  const context = page.context();
  const report = [];
  const LANG_BUTTON = { ur: 'اردو', en: 'English' };
  for (const [w, h] of [
    [1366, 768],
    [1024, 600],
  ]) {
    for (const lang of ['ur', 'en']) {
      const scenes = [
        ['collection-page', 'vouchers?collection', null],
        ['opening-offer-today-open', 'index?screen=account&closedopen', 'offer'],
        ['opening-offer-today-closed', 'index?screen=account&closednight', 'offer'],
      ];
      for (const [name, hash, kind] of scenes) {
        const p = await context.newPage();
        p.on('dialog', (d) => d.accept().catch(() => undefined));
        await p.setViewportSize({ width: w, height: h });
        try {
          await p.goto(BASE + hash);
          await p.waitForTimeout(2500);
          await p.getByRole('button', { name: LANG_BUTTON[lang], exact: true }).first().click();
          await p.waitForTimeout(900);
          let found = null;
          if (kind === 'offer') {
            const inputs = p.locator('input');
            await inputs.nth(2).fill('اکرم');
            await inputs.nth(3).fill('Akram');
            await inputs.nth(13).fill('5000');
            await p.keyboard.press('F10');
            await p.waitForTimeout(1500);
            found = await p.evaluate(() => [...document.querySelectorAll('[role="status"], [role="alert"], [data-tone]')].map((el) => (el.textContent ?? '').trim().slice(0, 140)).filter(Boolean));
          }
          const file = `${OUT}/${name}-${lang}-${w}x${h}.png`;
          await p.screenshot({ path: file });
          report.push({
            name,
            lang,
            size: `${w}x${h}`,
            sideways: await p.evaluate(() => document.documentElement.scrollWidth > document.documentElement.clientWidth + 1),
            found,
          });
        } catch (e) {
          report.push({ name, lang, size: `${w}x${h}`, error: String(e).slice(0, 200) });
        } finally {
          await p.close({ runBeforeUnload: false }).catch(() => undefined);
        }
      }
    }
  }
  return report;
}
