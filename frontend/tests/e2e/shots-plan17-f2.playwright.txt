// Screenshots of the screens of the second half of plan 17 (F2: جنرل, سرمایہ, the partner statement, the overview and the day summary) in both languages at the three window sizes,
// taken from the development page /dev/rokar with the Playwright tool. It is a developer tool: run
// it with the Vite dev server up (`pnpm --filter frontend dev`). It also measures sideways overflow
// (NFR-USE-15): the screen area, and every table or grid inside it, must not scroll sideways.
// Usage with the Playwright MCP tool: browser_run_code_unsafe with `filename` set to this file.
// (It is a snippet for that tool, not a module, so it is kept as .txt and the linter does not read
// it. The Windows check is spec 24.)
async (page) => {
  const OUT =
    '/Users/faisalhanif/MyProfile/My Projects /Fruit Mandi Project/frontend/tests/e2e/output/screenshots/plan17-f2';
  const BASE = 'http://localhost:5173/#/dev/rokar';
  const SIZES = [
    [1024, 600],
    [1366, 768],
    [1920, 1080],
  ];
  const settle = (ms) => page.waitForTimeout(ms);
  await page.addInitScript(() => {
    const add = window.addEventListener.bind(window);
    window.addEventListener = (type, ...rest) => {
      if (type !== 'beforeunload') add(type, ...rest);
    };
  });
  const amount = () => page.locator('main input[inputmode="numeric"]').first();
  const chooseSelect = async (index, optionIndex) => {
    await page.locator('main button[role="combobox"]').nth(index).click();
    await settle(150);
    await page.getByRole('option').nth(optionIndex).click();
    await settle(150);
  };
  const pickPicker = async (text) => {
    const box = page.locator('main input[role="combobox"]').first();
    await box.click();
    await box.fill(text);
    await settle(300);
    await page.keyboard.press('Enter');
    await settle(300);
  };
  const cell = (row, col) => page.locator(`main .ag-row[row-index="${row}"] .ag-cell[col-id="${col}"]`);
  const typeInCell = async (row, col, text) => {
    await cell(row, col).click();
    await page.keyboard.type(text, { delay: 30 });
    await settle(400);
  };
  const SCENARIOS = {
    'general-empty': { query: '?screen=general', run: async () => {} },
    'general-balanced': {
      query: '?screen=general',
      run: async () => {
        await typeInCell(0, 'account', 'elec');
        await page.keyboard.press('Enter');
        await settle(300);
        await page.keyboard.type('3000');
        await page.keyboard.press('Enter');
        await page.keyboard.press('Enter');
        await page.keyboard.press('Enter');
        await settle(200);
        await typeInCell(1, 'account', 'cash');
        await page.keyboard.press('Enter');
        await settle(300);
        await page.keyboard.press('Enter');
        await page.keyboard.type('3000');
        await page.keyboard.press('Enter');
        await settle(700);
      },
    },
    'general-not-balanced': {
      query: '?screen=general',
      run: async () => {
        await typeInCell(0, 'account', 'elec');
        await page.keyboard.press('Enter');
        await settle(300);
        await page.keyboard.type('3000');
        await page.keyboard.press('Enter');
        await settle(700);
      },
    },
    'capital-empty': { query: '?screen=capital&type=capital_in', run: async () => {} },
    'capital-out': {
      query: '?screen=capital&type=capital_out&partner=2',
      run: async () => {
        await amount().fill('10000');
        await settle(600);
      },
    },
    'statement-b': { query: '?screen=statement&partner=2', run: async () => { await settle(500); } },
    'statement-a-trip': { query: '?screen=statement&partner=1', run: async () => { await settle(500); } },
    'overview': { query: '?screen=overview', run: async () => { await settle(500); } },
    'day-summary': { query: '?screen=day', run: async () => { await settle(700); } },
  };
  const result = [];
  for (const [name, scenario] of Object.entries(SCENARIOS)) {
    for (const language of ['ur', 'en']) {
      for (const [width, height] of SIZES) {
        await page.setViewportSize({ width, height });
        await page.goto('about:blank');
        await page.goto(`${BASE}${scenario.query}`);
        await page.reload();
        await settle(900);
        if (language === 'en') {
          await page.keyboard.press('Alt+KeyL');
          await settle(500);
        }
        await scenario.run();
        const overflow = await page.evaluate(() => {
          const main = document.querySelector('main');
          const wide = [...document.querySelectorAll('main *')].filter(
            (el) => el.scrollWidth > el.clientWidth + 1 && getComputedStyle(el).overflowX === 'visible' && el.clientWidth > 0,
          );
          return {
            main: main ? main.scrollWidth - main.clientWidth : -1,
            page: document.documentElement.scrollWidth - document.documentElement.clientWidth,
            wide: wide.length,
          };
        });
        const file = `${OUT}/${name}-${language}-${width}x${height}.png`;
        await page.screenshot({ path: file, fullPage: false });
        result.push(`${name} ${language} ${width}x${height} overflow main=${overflow.main} page=${overflow.page} wide=${overflow.wide}`);
      }
    }
  }
  return result.join('\n');
}
