// Screenshots of the report prints on the shared print layout (plan 30, 17e): the print preview of R5,
// R37, R31, R34, R18 and the analytics views K and J, and the slips of all the credit buyers of R5, in
// both languages on A4 and A5, taken from the development pages /dev/reports/<id> and
// /dev/analytics/<view>. The print is the shared PrintRoot with the ONE PrintHeader (firm, contacts,
// trade mark, title, filters) and the footer of the backend's `reports_print_header`. It is a developer
// tool: run it with the Vite dev server up on a free port (`pnpm --filter frontend exec vite --port
// 5188 --strictPort --host 127.0.0.1`) in its own headless Chrome with a throwaway profile (a small
// Node runner that evaluates this text with playwright-core). It measures sideways overflow of the
// page and of every sheet (NFR-USE-15), counts the sheets, and checks that no digit outside 0-9 is on
// the page and that no raw markup, message ID or error code shows. The snippet writes the picture of
// the document itself with an absolute path from inside the page code. (A snippet for that tool, not a
// module, so it is kept as .txt and the linter does not read it. The Windows check is spec 53.)
//
// [route, flags, name of the shot, the button that opens the print].
async (page) => {
  const OUT =
    '/Users/faisalhanif/MyProfile/My Projects /Fruit Mandi Parallel/d5/frontend/tests/e2e/output/screenshots/plan30';
  const HOST = 'http://127.0.0.1:5188/#/dev/';
  const SIZES = [
    [1024, 600],
    [1366, 768],
    [1920, 1080],
  ];
  const PREVIEW = /^(Print preview|پرنٹ پر نظر)/;
  const SLIPS = /^(Print all slips|تمام پرچیاں پرنٹ کریں)/;
  const SHOTS = [
    ['reports/R5', '', 'print-r5', PREVIEW],
    ['reports/R5', '', 'print-r5-slips', SLIPS],
    ['reports/R37', '', 'print-r37', PREVIEW],
    ['reports/R34', '?fruit', 'print-r34', PREVIEW],
    ['reports/R18', '', 'print-r18', PREVIEW],
    ['analytics/K', '', 'print-k', PREVIEW],
    ['analytics/J', '', 'print-j', PREVIEW],
  ];
  const PAPERS = [
    ['a4', /A4|کاغذ A4/],
    ['a5', /A5|کاغذ A5/],
  ];
  const results = [];
  for (const [width, height] of SIZES) {
    await page.setViewportSize({ width, height });
    for (const [route, flags, name, button] of SHOTS) {
      for (const lang of ['ur', 'en']) {
        try {
          await page.goto(`${HOST}${route}${flags}`);
          await page.reload({ waitUntil: 'load', timeout: 60000 });
          await page.waitForSelector('h1, [role="alert"]', { timeout: 45000 });
          for (let step = 0; step < 2; step += 1) {
            const now = await page.evaluate(() => document.documentElement.lang);
            if (now === lang) break;
            await page.keyboard.press('Alt+l');
            await page.waitForTimeout(250);
          }
          await page.waitForSelector('.ag-row', { timeout: 45000 });
          await page.getByRole('button', { name: button }).first().click();
          await page.waitForSelector('.print-sheet', { timeout: 30000 });
          await page.waitForTimeout(700);
          for (const [paper, paperName] of PAPERS) {
            if (width !== 1366 && paper === 'a5') continue;
            await page.getByRole('button', { name: paperName }).first().click();
            await page.waitForTimeout(900);
            await page.waitForSelector('.print-sheet', { timeout: 30000 });
            const o = await page.evaluate(() => {
              const root = document.documentElement;
              const sheets = Array.from(document.querySelectorAll('.print-sheet'));
              const sheetX = sheets.map((n) => n.scrollWidth - n.clientWidth);
              const text = document.querySelector('.print-doc')?.textContent ?? '';
              const eastern = /[۰-۹٠-٩]/.test(text);
              const raw = /report-col-|report-r\d+-|analytics-col-|analytics-view-|print-fact|ERR_|<num>/.test(
                text,
              );
              return [
                root.lang,
                root.dir,
                'pageX=' + (root.scrollWidth - window.innerWidth),
                'sheets=' + sheets.length,
                'sheetX=' + sheetX.join(','),
                'eastern=' + eastern,
                'raw=' + raw,
              ].join(' | ');
            });
            await page
              .locator('.print-doc')
              .first()
              .screenshot({ path: `${OUT}/${name}-${paper}-${lang}-${width}x${height}.png` });
            results.push(`${name} ${paper} ${lang} ${width} ${o}`);
          }
        } catch (e) {
          results.push(`${name} ${lang} ${width} FAILED ${String(e).slice(0, 140)}`);
        }
      }
    }
  }
  return results.join('\n');
};
