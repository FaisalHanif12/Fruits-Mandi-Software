// Screenshots of the voucher screens in both languages at the three window sizes (plan 15), taken
// from the development page /dev/vouchers with the Playwright tool. It is a developer tool: run it
// with the Vite dev server up (`pnpm --filter frontend dev`). It also measures sideways overflow
// (NFR-USE-15): the screen area, and every table or grid inside it, must not scroll sideways.
// Usage with the Playwright MCP tool: browser_run_code_unsafe with `filename` set to this file.
// (It is a snippet for that tool, not a module: `page` has no types here, so it is kept as .txt and
// the linter does not read it. The Windows check is the specs 16 and 17.)
async (page) => {
  const OUT =
    '/Users/faisalhanif/MyProfile/My Projects /Fruit Mandi Project/frontend/tests/e2e/output/screenshots/plan15';
  const BASE = 'http://localhost:5173/#/dev/vouchers';
  const SIZES = [
    [1024, 600],
    [1366, 768],
    [1920, 1080],
  ];
  const ONLY = null;
  const settle = (ms) => page.waitForTimeout(ms);
  // (The page's own "leave?" listener is switched off before every load, so a reload never waits.)
  await page.addInitScript(() => {
    const add = window.addEventListener.bind(window);
    window.addEventListener = (type, ...rest) => {
      if (type !== 'beforeunload') add(type, ...rest);
    };
  });
  const first = () => page.locator('main input[role="combobox"]').first();
  const amount = () => page.locator('main input[inputmode="numeric"]').first();
  async function pick(text) {
    await first().click();
    await first().fill(text);
    await settle(250);
    await page.keyboard.press('Enter');
    await settle(250);
  }
  const SCENARIOS = {
    'receipt-empty': { query: '?type=receipt', run: async () => {} },
    'receipt-buyer': {
      query: '?type=receipt',
      run: async () => {
        await pick('03040001');
        await amount().fill('10000');
        await settle(500);
      },
    },
    'payment-seller-over': {
      query: '?type=payment',
      run: async () => {
        await pick('030110001');
        await amount().fill('5000');
        await settle(500);
      },
    },
    'payment-buyer-lent': {
      query: '?type=payment',
      run: async () => {
        await pick('03040001');
        await amount().fill('3000');
        await settle(500);
      },
    },
    'advance-cash-below-zero': {
      query: '?type=advance',
      run: async () => {
        await pick('030110001');
        await amount().fill('60000');
        await settle(500);
      },
    },
    'edit-saved': { query: '?type=receipt&id=501', run: async () => settle(600) },
    'view-cancelled': { query: '?type=receipt&id=503', run: async () => settle(600) },
    'collection-round': {
      query: '?collection',
      run: async () => {
        await settle(600);
        const cells = page.locator('.ag-row[row-index="0"] [col-id="received"]');
        await cells.click();
        await page.keyboard.type('2880');
        await page.keyboard.press('Enter');
        await settle(300);
      },
    },
    'waiting-list': {
      query: '?type=receipt',
      run: async () => {
        await pick('03040001');
        await amount().fill('10000');
        await settle(400);
        await page.keyboard.press('F2');
        await settle(500);
        await pick('03040002');
        await amount().fill('5000');
        await settle(400);
        await page.keyboard.press('F2');
        await settle(500);
        await pick('03040001');
        await amount().fill('2880');
        await settle(500);
      },
    },
    'waiting-list-bottom': {
      query: '?type=receipt',
      run: async () => {
        await pick('03040001');
        await amount().fill('10000');
        await settle(400);
        await page.keyboard.press('F2');
        await settle(500);
        await pick('03040002');
        await amount().fill('5000');
        await settle(400);
        await page.keyboard.press('F2');
        await settle(500);
        await page.evaluate(() => document.querySelector('main')?.scrollTo(0, 99999));
        await settle(300);
      },
    },
    'cancel-dialog': {
      query: '?type=receipt',
      run: async () => {
        await settle(600);
        await page.locator('.ag-row[row-id="501"] [col-id="number"]').click();
        await page.locator('main button.text-due').click();
        await settle(400);
      },
    },
    'print-box': {
      query: '?type=receipt',
      run: async () => {
        await settle(600);
        await page.locator('.ag-row[row-id="501"] [col-id="number"]').click();
        await settle(300);
        await page.keyboard.press('F9');
        await settle(800);
      },
    },
    'print-preview': {
      query: '?type=receipt',
      run: async () => {
        await settle(600);
        await page.locator('.ag-row[row-id="501"] [col-id="number"]').click();
        await settle(300);
        await page.keyboard.press('F9');
        await settle(800);
        const buttons = page.locator('[role="dialog"] button');
        await buttons.first().waitFor({ timeout: 3000 });
        const n = await buttons.count();
        await buttons.nth(n - 2).click();
        await settle(900);
      },
    },
  };
  const report = [];
  for (const [name, scenario] of Object.entries(SCENARIOS)) {
    if (ONLY !== null && !ONLY.includes(name)) continue;
    for (const language of ['ur', 'en']) {
      for (const [w, h] of SIZES) {
        await page.setViewportSize({ width: w, height: h });
        await page.goto(`${BASE}${scenario.query}`);
        await page.reload();
        await settle(700);
        if (language === 'en') await page.keyboard.press('Alt+L');
        await settle(300);
        try {
          await scenario.run();
        } catch (failure) {
          report.push(`${name} ${language} ${w}x${h} FAILED ${String(failure).slice(0, 120)}`);
        }
        const overflow = await page.evaluate(() => {
          const found = [];
          const main = document.querySelector('main');
          if (main && main.scrollWidth > main.clientWidth + 1)
            found.push(`main ${main.scrollWidth}>${main.clientWidth}`);
          for (const el of document.querySelectorAll(
            'main .overflow-x-auto, main .ag-body-viewport, main .ag-center-cols-viewport, main .ag-body-horizontal-scroll-viewport',
          )) {
            if (el.scrollWidth > el.clientWidth + 1)
              found.push(
                `${el.className.toString().slice(0, 40)} ${el.scrollWidth}>${el.clientWidth}`,
              );
          }
          return found;
        });
        await page.screenshot({ path: `${OUT}/${name}__${language}__${w}x${h}.png` });
        report.push(
          `${name} ${language} ${w}x${h}${overflow.length ? ' OVERFLOW ' + overflow.join('; ') : ''}`,
        );
      }
    }
  }
  return report.join('\n');
}
